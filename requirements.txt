# Dipendenze di Orchestra.
#
# `ddgs` è l'unica che porta funzioni nuove (la ricerca sul web senza
# chiave). Il resto è standard library. Se manca, Orchestra funziona
# comunque: i tool del web lo dicono invece di morire.
ddgs>=9.0

# `Pillow` serve al montaggio deterministico delle immagini (`immagine`,
# azione "modifica": testo, logo, ritaglio su una foto esistente). Senza,
# quel tool dice cosa installare e la generazione continua lo stesso.
Pillow>=10.1

# `pypdf` legge il testo dei PDF: un allegato PDF con testo costa zero
# token e arriva esatto, invece di essere dichiarato "scansionato" (era
# quello che succedeva senza questa riga: il messaggio incolpava il file
# mentre il guasto era la libreria mancante). Senza, Orchestra funziona
# lo stesso e dice quale comando manca.
pypdf>=5.0
